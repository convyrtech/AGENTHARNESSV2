Developer experience

Подключение — одна строка.

Любой официальный SDK. Любой язык. Просто замените base_url на https://api.onekey.dev/v1.

curl https://api.onekey.dev/v1/chat/completions \
  -H "Authorization: Bearer $ONEKEY_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "claude-sonnet-4",
    "messages": [{"role": "user", "content": "Привет"}],
    "stream": true
  }'
import os
from openai import OpenAI

client = OpenAI(
    base_url="https://api.onekey.dev/v1",
    api_key=os.environ["ONEKEY_API_KEY"],
)

stream = client.chat.completions.create(
    model="gpt-4o",  # claude-sonnet-4, gemini-2.5-pro, deepseek-v3
    messages=[{"role": "user", "content": "Привет"}],
    stream=True,
)

for chunk in stream:
    print(chunk.choices[0].delta.content or "", end="")
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "https://api.onekey.dev/v1",
  apiKey: process.env.ONEKEY_API_KEY,
});

const stream = await client.chat.completions.create({
  model: "gemini-2.5-pro",
  messages: [{ role: "user", content: "Привет" }],
  stream: true,
});

for await (const chunk of stream) {
  process.stdout.write(chunk.choices[0]?.delta?.content ?? "");
}
client := openai.NewClient(
    option.WithBaseURL("https://api.onekey.dev/v1"),
    option.WithAPIKey(os.Getenv("ONEKEY_API_KEY")),
)

stream, err := client.CreateChatCompletionStream(ctx,
    openai.ChatCompletionRequest{
        Model:    "deepseek-v3",
        Messages: []openai.ChatCompletionMessage{
            {Role: "user", Content: "Привет"},
        },
    },
)
if err != nil {
    log.Fatal(err)
}
defer stream.Close()

for {
    res, err := stream.Recv()
    if errors.Is(err, io.EOF) {
        break
    }
    fmt.Print(res.Choices[0].Delta.Content)
}

Коротко о главном

Мы работаем с провайдерами напрямую и закупаем объёмы оптом — отсюда цена ниже розничной. Вы получаете собственный ключ onekey и стабильный доступ к моделям. Никаких «общих» аккаунтов и очередей: каждый запрос идёт через ваш ключ с собственными лимитами.

Шлюз видит ошибку 429/5xx и повторяет запрос, а при повторной неудаче — уводит его в резервную модель, которую вы указали в конфиге. Ответ клиенту приходит в том же формате, поэтому приложение не падает. В логах видно, какой запрос был переадресован.

По токенам — вход и выход считаются отдельно, как у провайдеров. Цена фиксируется в момент запроса и не меняется задним числом. В дашборде видна стоимость каждого вызова с разбивкой по модели и проекту.
