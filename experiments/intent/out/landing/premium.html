Простой как 2×2 API

Совместимость с OpenAI API. Меняете endpoint — всё работает.

# Вместо OpenAI:
openai.api_base = "https://api.openai.com/v1"
# Используйте NeuralHub:
openai.api_base = "https://api.neuralhub.dev/v1"
openai.api_key = "your-neuralhub-key"
# Переключение между моделями:
model = "gpt-4-turbo" # OpenAI
model = "claude-3.5-sonnet" # Anthropic
model = "gemini-pro" # Google
model = "deepseek-v3" # DeepSeek
